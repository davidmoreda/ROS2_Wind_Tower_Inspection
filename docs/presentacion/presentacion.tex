\section{Contexto y objetivo}

\begin{frame}{Contexto y objetivo del proyecto}
  % TODO P1: por que la inspeccion autonoma de torres eolicas
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Las turbinas eolicas operan en condiciones extremas y requieren \highlight{inspeccion periodica} de defectos en las palas y la torre.
        \item Hoy la inspeccion la hace personal en altura con drones y camaras manuales.
        \item Nuestra propuesta: un \highlight{robot terrestre con brazo manipulador} que inspecciona desde dentro del tubo de la torre.
      \end{itemize}
      \vspace{0.5em}
      \textbf{Objetivo:} integrar navegacion autonoma, manipulacion, percepcion y comportamiento en un unico sistema simulado en Gazebo.
    \end{column}
    \begin{column}{0.45\textwidth}
      \fig[\textwidth]{contexto_torre.png}
    \end{column}
  \end{columns}
\end{frame}

\section{Navegacion autonoma (NAV2)}

\begin{frame}{Stack de navegacion: SLAM + EKF + AMCL}
  % TODO P2: explicar el stack
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \highlight{SLAM Toolbox} para mapeo offline del entorno.
        \item \highlight{EKF} (\texttt{robot\_localization}) fusiona IMU + odometria $\to$ TF \texttt{odom -> base\_link}.
        \item \highlight{AMCL} localiza globalmente sobre el mapa $\to$ TF \texttt{map -> odom}.
        \item \texttt{nav2\_bringup} orquesta planner (NavFn), controller (RPP), BT navigator y recovery.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \fig[\textwidth]{nav2_rviz.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Tuning del controller: rampa de 13°}
  % TODO P2: contar la historia del tuning (rampa, anti-bandazos, slip).

    \emph{"Aceleraciones suaves para no perder traccion."}
  \end{block}

  \begin{tabular}{lrr}

    \texttt{smoother.max\_accel}    & 2.5  & \textbf{0.75} \\
    \texttt{rotate\_to\_heading\_min\_angle} & 45° & \textbf{60°} \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Deteccion de defectos con YOLO}
  % TODO P3: YOLO + dataset
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
